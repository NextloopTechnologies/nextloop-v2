# robots.txt has no inheritance: a crawler obeys the single most specific group
# that matches it and ignores every other group entirely. The named groups below
# previously carried only "Allow: /", so the Disallow rules in the "*" group did
# not apply to Googlebot, Bingbot or any of the AI crawlers — all of them were
# free to crawl /admin/. Each group now repeats the rules it needs.

# /payload-api/ is Payload's REST API. The lead, resume and user collections
# already answer 403 to an anonymous caller, so this is not what keeps them
# private — access control is. It is here because the *public* collections do
# answer 200, and a crawler indexing /payload-api/blogs/ would be indexing the
# same content twice, once as JSON with no canonical pointing home.

Allow: /
Disallow: /admin/
Disallow: /config/
Disallow: /api/private/
Disallow: /payload-api/
